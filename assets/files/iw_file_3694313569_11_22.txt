<script>
  const gameArea = document.getElementById('gameArea');
  const score1El = document.getElementById('score1');
  const score2El = document.getElementById('score2');
  const timerEl = document.getElementById('timer');
  const currentPlayerEl = document.getElementById('currentPlayer');
  const messageEl = document.getElementById('message');
  const restartBtn = document.getElementById('restartBtn');
  const infoBtn = document.getElementById('infoBtn');
  const infoOverlay = document.getElementById('infoOverlay');
  const closeInfoBtn = document.getElementById('closeInfoBtn');
  const shootSound = document.getElementById('shootSound');
  const hitSound = document.getElementById('hitSound');

  let score = [0, 0];
  let currentPlayer = 0; // 0 = 玩家1, 1 = 玩家2
  let timeLeft = 60;
  let gameInterval;
  let targetMoveInterval;
  let targets = [];
  const maxTargets = 5;

  // 產生一個 Roblox 方塊風格目標元素
  function createTarget() {
    const target = document.createElement('div');
    target.classList.add('target');
    // Roblox風格配色(紅色系方塊)
    // 文字顯示分數加成(可擴充)
    target.textContent = '■';
    // 初始隨機位置
    const x = Math.random() * (gameArea.clientWidth - 60);
    const y = Math.random() * (gameArea.clientHeight - 60);
    target.style.left = x + 'px';
    target.style.top = y + 'px';
    target.dataset.point = 1; // 每個目標分數
    target.dataset.id = Date.now() + Math.random();

    // 點擊事件
    target.addEventListener('click', () => {
      shootSound.currentTime = 0;
      shootSound.play();
      hitSound.currentTime = 0;
      hitSound.play();
      addScore(currentPlayer, parseInt(target.dataset.point));
      animateHit(target);
      removeTarget(target);
      switchPlayer();
    });

    gameArea.appendChild(target);
    targets.push(target);
  }

  // 動畫效果：簡單放大縮小模擬爆炸
  function animateHit(target) {
    target.style.transition = 'transform 0.3s ease';
    target.style.transform = 'scale(1.5)';
    setTimeout(() => {
      if (target.parentElement) target.style.transform = 'scale(0)';
    }, 250);
  }

  // 移除目標
  function removeTarget(target) {
    targets = targets.filter(t => t !== target);
    if (target.parentElement) target.parentElement.removeChild(target);
  }

  // 增加分數
  function addScore(playerIdx, points) {
    score[playerIdx] += points;
    updateScoreDisplay();
  }

  // 更新分數顯示
  function updateScoreDisplay() {
    score1El.textContent = score[0];
    score2El.textContent = score[1];
  }

  // 切換玩家
  function switchPlayer() {
    currentPlayer = 1 - currentPlayer;
    currentPlayerEl.textContent = currentPlayer + 1;
  }

  // 遊戲倒數計時
  function startTimer() {
    timerEl.textContent = timeLeft;
    gameInterval = setInterval(() => {
      timeLeft--;
      timerEl.textContent = timeLeft;
      if (timeLeft <= 0) {
        endGame();
      }
    }, 1000);
  }

  // 遊戲結束顯示結果
  function endGame() {
    clearInterval(gameInterval);
    clearInterval(targetMoveInterval);
    removeAllTargets();

    let resultMsg = '';
    if (score[0] > score[1]) {
      resultMsg = '玩家 1 獲勝！🎉';
    } else if (score[1] > score[0]) {
      resultMsg = '玩家 2 獲勝！🎉';
    } else {
      resultMsg = '平手！🤝';
    }
    messageEl.textContent = `遊戲結束！${resultMsg} 最終分數：玩家1 ${score[0]} 分，玩家2 ${score[1]} 分`;
  }

  // 移除所有目標
  function removeAllTargets() {
    targets.forEach(t => {
      if (t.parentElement) t.parentElement.removeChild(t);
    });
    targets = [];
  }

  // 隨機移動目標
  function moveTargets() {
    targets.forEach(target => {
      let x = parseFloat(target.style.left);
      let y = parseFloat(target.style.top);
      // 隨機方向移動，範圍限制在遊戲區域內
      x += (Math.random() - 0.5) * 30;
      y += (Math.random() - 0.5) * 30;
      x = Math.min(Math.max(0, x), gameArea.clientWidth - 60);
      y = Math.min(Math.max(0, y), gameArea.clientHeight - 60);
      target.style.left = x + 'px';
      target.style.top = y + 'px';
    });
  }

  // 遊戲主循環：持續維持目標數量與移動
  function gameLoop() {
    if (targets.length < maxTargets) {
      createTarget();
    }
  }

  // 重新開始遊戲
  function restartGame() {
    clearInterval(gameInterval);
    clearInterval(targetMoveInterval);
    removeAllTargets();
    score = [0, 0];
    currentPlayer = 0;
    timeLeft = 60;
    updateScoreDisplay();
    currentPlayerEl.textContent = '1';
    messageEl.textContent = '';
    startTimer();
    targetMoveInterval = setInterval(moveTargets, 500);
    gameLoopInterval = setInterval(gameLoop, 1000);
  }

  // 監聽重新開始按鈕
  restartBtn.addEventListener('click', () => {
    restartGame();
  });

  // 遊戲說明彈窗開關
  infoBtn.addEventListener('click', () => {
    infoOverlay.style.display = 'flex';
  });
  closeInfoBtn.addEventListener('click', () => {
    infoOverlay.style.display = 'none';
  });

  // 初始化遊戲
  let gameLoopInterval;
  restartGame();

</script>

</body>
